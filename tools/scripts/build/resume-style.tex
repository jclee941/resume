% Clean, print-friendly resume styling for Pandoc + XeLaTeX.
% Keep deterministic: no dates, timestamps, random marks, or runtime values.
\usepackage{xcolor}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage{parskip}
\usepackage{fancyhdr}

\definecolor{ResumeSlate}{HTML}{1F2933}
\definecolor{ResumeText}{HTML}{2F3A45}
\definecolor{ResumeMuted}{HTML}{6B7280}
\definecolor{ResumeRule}{HTML}{D5DCE3}
\definecolor{ResumeAccent}{HTML}{5AA9B8}
\definecolor{ResumeAccentDark}{HTML}{357C8A}

\AtBeginDocument{%
  \color{ResumeText}%
  \pagestyle{fancy}%
  \thispagestyle{empty}%
  \fancyhf{}%
  \fancyhead[L]{\footnotesize\color{ResumeMuted}이재철}%
  \fancyhead[R]{\footnotesize\color{ResumeMuted}\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

\setlength{\headheight}{14pt}
\setlength{\columnsep}{18pt}
\setlength{\parindent}{0pt}
\setlength{\parskip}{2pt plus 0.6pt minus 0.6pt}
\setlength{\emergencystretch}{2em}

\setlist[itemize]{%
  leftmargin=1.25em,
  itemsep=1pt plus 0.5pt minus 0.5pt,
  topsep=2pt plus 0.5pt minus 0.5pt,
  parsep=0pt,
  partopsep=0pt
}
\setlist[enumerate]{%
  leftmargin=1.35em,
  itemsep=1pt plus 0.5pt minus 0.5pt,
  topsep=2pt plus 0.5pt minus 0.5pt,
  parsep=0pt,
  partopsep=0pt
}

% Pandoc maps the resume's markdown H1 (# 이재철) to \section and H2
% sections (## 연락처, ## 경력사항, ...) to \subsection. Style both levels
% intentionally so the H1 acts as a clean name header and H2 sections carry
% the restrained divider rule.
\titleformat{\section}
  {\Large\bfseries\color{ResumeSlate}}
  {}
  {0pt}
  {}
\titlespacing*{\section}{0pt}{0.05em}{0.35em}

\titleformat{\subsection}
  {\normalsize\bfseries\color{ResumeSlate}}
  {}
  {0pt}
  {}
  [\vspace{0.15em}{\color{ResumeAccent}\titlerule[0.45pt]}]
\titlespacing*{\subsection}{0pt}{0.75em plus 0.15em minus 0.1em}{0.35em}

\titleformat{\subsubsection}
  {\small\bfseries\color{ResumeAccentDark}}
  {}
  {0pt}
  {}
\titlespacing*{\subsubsection}{0pt}{0.55em plus 0.1em minus 0.1em}{0.2em}

\titleformat{\paragraph}[runin]
  {\normalsize\bfseries\color{ResumeSlate}}
  {}
  {0pt}
  {}
  [\hspace{0.35em}]
\titlespacing*{\paragraph}{0pt}{0.45em}{0pt}
